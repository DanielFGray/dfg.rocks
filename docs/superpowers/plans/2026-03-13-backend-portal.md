# dfg.rocks Backend & Student Portal — Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Bun + Effect-TS backend with contact form handling, admin dashboard, and student portal using Fibrae SPA.

**Architecture:** Separate project at `~/build/dfg-portal/`. Effect HttpApi on Bun.serve, Postgres via Graphile Migrate with RLS, Valkey sessions, Fibrae SPA served as static shell. Follows postgres.garden patterns.

**Tech Stack:** Bun, Effect-TS, @effect/platform HttpApi, @effect/sql-kysely, PostgreSQL 17, Valkey, Graphile Migrate, Fibrae, Vite, Resend, Docker Compose, Caddy

**Spec:** `docs/superpowers/specs/2026-03-13-backend-portal-design.md`

```typescript
#!/usr/bin/env bun
import crypto from "crypto";
import fs from "fs";
import path from "path";

const DB_NAME = "dfg_portal";
const DB_HOST = "localhost";
const DB_PORT = "5433";
const VALKEY_PORT = "6380";

function genPassword(len = 32) {
  return crypto.randomBytes(len).toString("base64url");
}

function genSecret(len = 64) {
  return crypto.randomBytes(len).toString("hex");
}

const rootPassword = "postgres"; // dev default
const ownerPassword = genPassword();
const authPassword = genPassword();
const shadowPassword = genPassword();

const env = `
NODE_ENV=development

ROOT_DATABASE_USER=postgres
ROOT_DATABASE_PASSWORD=${rootPassword}

DATABASE_HOST=${DB_HOST}
DATABASE_PORT=${DB_PORT}
DATABASE_NAME=${DB_NAME}

DATABASE_OWNER=${DB_NAME}
DATABASE_OWNER_PASSWORD=${ownerPassword}
DATABASE_AUTHENTICATOR=${DB_NAME}_authenticator
DATABASE_AUTHENTICATOR_PASSWORD=${authPassword}
DATABASE_VISITOR=${DB_NAME}_visitor
SHADOW_DATABASE_PASSWORD=${shadowPassword}

ROOT_DATABASE_URL=postgres://postgres:${rootPassword}@${DB_HOST}:${DB_PORT}/template1
DATABASE_URL=postgres://${DB_NAME}:${ownerPassword}@${DB_HOST}:${DB_PORT}/${DB_NAME}
AUTH_DATABASE_URL=postgres://${DB_NAME}_authenticator:${authPassword}@${DB_HOST}:${DB_PORT}/${DB_NAME}
SHADOW_DATABASE_URL=postgres://${DB_NAME}_shadow:${shadowPassword}@${DB_HOST}:${DB_PORT}/${DB_NAME}_shadow

PORT=3001
VITE_ROOT_URL=http://localhost:3001
SECRET=${genSecret()}

VALKEY_URL=redis://${DB_HOST}:${VALKEY_PORT}

RESEND_API_KEY=
ADMIN_EMAIL=danielfgray@gmail.com
`.trim();

const envPath = path.join(import.meta.dir, "..", ".env");
if (fs.existsSync(envPath) && !process.env.NOCONFIRM) {
  console.log(".env already exists, skipping (set NOCONFIRM=1 to overwrite)");
```

### Task 4: Database setup script

**Files:**
- Create: `~/build/dfg-portal/scripts/dbSetup.ts`

- [ ] **Step 1: Write dbSetup.ts**

Adapt from postgres.garden. Creates roles + databases.

```typescript
#!/usr/bin/env bun
import pg from "pg";

const env = (k: string) => {
  const v = process.env[k];
  if (!v) throw new Error(`Missing env: ${k}`);
  return v;
};

async function main() {
  const client = new pg.Client({ connectionString: env("ROOT_DATABASE_URL") });
  await client.connect();

  const dbName = env("DATABASE_NAME");
  const owner = env("DATABASE_OWNER");
  const ownerPw = env("DATABASE_OWNER_PASSWORD");
  const auth = env("DATABASE_AUTHENTICATOR");
  const authPw = env("DATABASE_AUTHENTICATOR_PASSWORD");
  const visitor = env("DATABASE_VISITOR");
  const shadowPw = env("SHADOW_DATABASE_PASSWORD");

  const sql = (s: string) => client.query(s).catch((e) => {
    if (!e.message.includes("does not exist")) throw e;
  });

  // Drop existing
  for (const db of [dbName, `${dbName}_shadow`, `${dbName}_test`]) {
    await sql(`DROP DATABASE IF EXISTS "${db}"`);
  }
  for (const role of [owner, auth, visitor, `${dbName}_shadow`]) {
    await sql(`DROP ROLE IF EXISTS "${role}"`);
  }

  // Create roles
  await client.query(`CREATE ROLE "${owner}" LOGIN PASSWORD '${ownerPw}'`);
  await client.query(`CREATE ROLE "${auth}" LOGIN PASSWORD '${authPw}' NOINHERIT`);
  await client.query(`CREATE ROLE "${visitor}"`);
  await client.query(`CREATE ROLE "${dbName}_shadow" LOGIN PASSWORD '${shadowPw}' SUPERUSER`);
  await client.query(`GRANT "${visitor}" TO "${auth}"`);

  // Create databases
  await client.query(`CREATE DATABASE "${dbName}" OWNER "${owner}"`);
  await client.query(`CREATE DATABASE "${dbName}_shadow" OWNER "${dbName}_shadow"`);
  await client.query(`CREATE DATABASE "${dbName}_test" OWNER "${owner}"`);

  console.log(`Created databases and roles for ${dbName}`);
  await client.end();
}

main().catch((e) => { console.error(e); process.exit(1); });
```

```sql
-- 0001-reset
DROP SCHEMA IF EXISTS app_public CASCADE;
DROP SCHEMA IF EXISTS app_private CASCADE;

-- 0010-schemas
CREATE SCHEMA app_public;
CREATE SCHEMA app_private;
GRANT USAGE ON SCHEMA app_public TO :DATABASE_VISITOR;

-- 0020-common-triggers
CREATE FUNCTION app_private.tg__timestamps() RETURNS trigger AS $$
BEGIN
  NEW.created_at = (CASE WHEN TG_OP = 'INSERT' THEN NOW() ELSE OLD.created_at END);
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql VOLATILE;

-- 0100-user-role-enum
CREATE TYPE app_public.user_role AS ENUM ('admin', 'student');

-- 0110-users
CREATE TABLE app_public.users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  username CITEXT NOT NULL UNIQUE,
  name TEXT NOT NULL DEFAULT '',
  email CITEXT NOT NULL UNIQUE,
  role app_public.user_role NOT NULL DEFAULT 'student',
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TRIGGER _timestamps
  BEFORE INSERT OR UPDATE ON app_public.users
  FOR EACH ROW EXECUTE FUNCTION app_private.tg__timestamps();

GRANT SELECT ON app_public.users TO :DATABASE_VISITOR;

-- RLS: users can read their own record, admins can read all
ALTER TABLE app_public.users ENABLE ROW LEVEL SECURITY;
CREATE POLICY select_own ON app_public.users FOR SELECT
  USING (id::text = current_setting('my.user_id', true) OR
         current_setting('my.user_role', true) = 'admin');

-- 0120-user-secrets
CREATE TABLE app_private.user_secrets (
  user_id UUID PRIMARY KEY REFERENCES app_public.users(id) ON DELETE CASCADE,
  password_hash TEXT,
  invite_token TEXT,
  invite_token_expires_at TIMESTAMPTZ,
  reset_password_token TEXT,
  reset_password_token_expires_at TIMESTAMPTZ
);

-- 0130-sessions
CREATE TABLE app_private.sessions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES app_public.users(id) ON DELETE CASCADE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  expires_at TIMESTAMPTZ NOT NULL
);

-- 0200-contact-submissions
CREATE TABLE app_public.contact_submissions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name TEXT NOT NULL,
  email TEXT NOT NULL,
  phone TEXT,
  instrument TEXT NOT NULL,
  format TEXT NOT NULL,
  message TEXT NOT NULL DEFAULT '',
  is_read BOOLEAN NOT NULL DEFAULT FALSE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Only admins can read submissions
ALTER TABLE app_public.contact_submissions ENABLE ROW LEVEL SECURITY;
CREATE POLICY admin_all ON app_public.contact_submissions
  USING (current_setting('my.user_role', true) = 'admin');
GRANT SELECT, UPDATE ON app_public.contact_submissions TO :DATABASE_VISITOR;
-- Insert allowed via authenticator role directly (no RLS for insert from API)
GRANT INSERT ON app_public.contact_submissions TO :DATABASE_AUTHENTICATOR;

-- 0300-courses
CREATE TABLE app_public.courses (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  title TEXT NOT NULL,
  description TEXT NOT NULL DEFAULT '',
  instrument TEXT NOT NULL DEFAULT '',
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TRIGGER _timestamps
  BEFORE INSERT OR UPDATE ON app_public.courses
  FOR EACH ROW EXECUTE FUNCTION app_private.tg__timestamps();

ALTER TABLE app_public.courses ENABLE ROW LEVEL SECURITY;
-- Students see courses they're enrolled in; admins see all
CREATE POLICY select_enrolled ON app_public.courses FOR SELECT
  USING (
    current_setting('my.user_role', true) = 'admin'
    OR id IN (
      SELECT course_id FROM app_public.enrollments
      WHERE student_id::text = current_setting('my.user_id', true)
    )
  );
GRANT SELECT ON app_public.courses TO :DATABASE_VISITOR;

-- 0310-modules
CREATE TABLE app_public.modules (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  course_id UUID NOT NULL REFERENCES app_public.courses(id) ON DELETE CASCADE,
  title TEXT NOT NULL,
  description TEXT,
  sort_order INTEGER NOT NULL DEFAULT 0
);

ALTER TABLE app_public.modules ENABLE ROW LEVEL SECURITY;
CREATE POLICY select_via_course ON app_public.modules FOR SELECT
  USING (
    current_setting('my.user_role', true) = 'admin'
    OR course_id IN (
      SELECT course_id FROM app_public.enrollments
      WHERE student_id::text = current_setting('my.user_id', true)
    )
  );
GRANT SELECT ON app_public.modules TO :DATABASE_VISITOR;

-- 0320-materials
CREATE TABLE app_public.materials (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  title TEXT NOT NULL,
  original_filename TEXT NOT NULL,
  file_path TEXT NOT NULL,
  mime_type TEXT NOT NULL DEFAULT 'application/octet-stream',
  file_size BIGINT NOT NULL DEFAULT 0,
  is_global BOOLEAN NOT NULL DEFAULT FALSE,
  uploaded_by UUID REFERENCES app_public.users(id) ON DELETE SET NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

ALTER TABLE app_public.materials ENABLE ROW LEVEL SECURITY;
-- Students see: global materials, materials in their enrolled courses, directly assigned
CREATE POLICY select_accessible ON app_public.materials FOR SELECT
  USING (
    current_setting('my.user_role', true) = 'admin'
    OR is_global
    OR id IN (
      SELECT mm.material_id FROM app_public.module_materials mm
      JOIN app_public.modules m ON m.id = mm.module_id
      JOIN app_public.enrollments e ON e.course_id = m.course_id
      WHERE e.student_id::text = current_setting('my.user_id', true)
    )
    OR id IN (
      SELECT material_id FROM app_public.student_materials
      WHERE student_id::text = current_setting('my.user_id', true)
    )
  );
GRANT SELECT ON app_public.materials TO :DATABASE_VISITOR;

  PRIMARY KEY (module_id, material_id)
);

ALTER TABLE app_public.module_materials ENABLE ROW LEVEL SECURITY;
CREATE POLICY select_via_module ON app_public.module_materials FOR SELECT

GRANT SELECT ON app_public.module_materials TO :DATABASE_VISITOR;

GRANT SELECT ON app_public.enrollments TO :DATABASE_VISITOR;

-- 0410-student-materials

  student_id UUID NOT NULL REFERENCES app_public.users(id) ON DELETE CASCADE,
  material_id UUID NOT NULL REFERENCES app_public.materials(id) ON DELETE CASCADE,
  assigned_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  PRIMARY KEY (student_id, material_id)
);

ALTER TABLE app_public.student_materials ENABLE ROW LEVEL SECURITY;
CREATE POLICY select_own ON app_public.student_materials FOR SELECT
  USING (
    current_setting('my.user_role', true) = 'admin'
    OR student_id::text = current_setting('my.user_id', true)
  );
GRANT SELECT ON app_public.student_materials TO :DATABASE_VISITOR;

-- 0500-auth-functions
CREATE FUNCTION app_private.login(identifier CITEXT, password TEXT)
RETURNS app_public.users AS $$
DECLARE
  v_user app_public.users;
  v_secrets app_private.user_secrets;
BEGIN
  SELECT u.* INTO v_user
    FROM app_public.users u
    WHERE u.email = identifier OR u.username = identifier;

  IF v_user IS NULL THEN
    -- Waste time to prevent timing attacks
    PERFORM pgcrypto.crypt(password, pgcrypto.gen_salt('bf'));
    RETURN NULL;
  END IF;

  SELECT * INTO v_secrets FROM app_private.user_secrets WHERE user_id = v_user.id;

  IF v_secrets.password_hash IS NULL OR
     v_secrets.password_hash <> pgcrypto.crypt(password, v_secrets.password_hash) THEN
    RETURN NULL;
  END IF;

  RETURN v_user;
END;
$$ LANGUAGE plpgsql STRICT SECURITY DEFINER;

CREATE FUNCTION app_private.set_password(v_user_id UUID, new_password TEXT)
RETURNS VOID AS $$
BEGIN
  UPDATE app_private.user_secrets
    SET password_hash = pgcrypto.crypt(new_password, pgcrypto.gen_salt('bf', 12))
    WHERE user_id = v_user_id;
END;
$$ LANGUAGE plpgsql STRICT SECURITY DEFINER;

CREATE FUNCTION app_private.create_user(
  v_username CITEXT,
  v_email CITEXT,
  v_name TEXT,
  v_role app_public.user_role,
  v_password TEXT DEFAULT NULL
)
RETURNS app_public.users AS $$
DECLARE
  v_user app_public.users;
BEGIN
  INSERT INTO app_public.users (username, email, name, role)
    VALUES (v_username, v_email, v_name, v_role)
    RETURNING * INTO v_user;

  INSERT INTO app_private.user_secrets (user_id)
    VALUES (v_user.id);

  IF v_password IS NOT NULL THEN
    PERFORM app_private.set_password(v_user.id, v_password);
  END IF;

  RETURN v_user;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
```

```typescript
const withRetry = <E, A, R>(layer: Layer.Layer<A, E, R>) =>
  Layer.retry(
    layer,
    Schedule.exponential("100 millis").pipe(
      Schedule.intersect(Schedule.recurs(10)),
```

Direct adaptation of postgres.garden's session service.

```typescript
import crypto from "crypto";
import { Effect } from "effect";
import { PgRootDB } from "../db.js";
import { valkey } from "../valkey.js";

export const SESSION_TTL_MS = 1000 * 60 * 60 * 24 * 30; // 30 days
const SESSION_TTL_SECONDS = 30 * 24 * 60 * 60;

function sessionKey(id: string) {
  return `dfg:session:${id}`;
}

function generateSecureRandomString(length = 32) {
  return crypto.randomBytes(length).toString("hex");
}

function hashSecret(secret: string) {
  return crypto.createHash("sha256").update(secret).digest();
}

function constantTimeEqual(a: Uint8Array, b: Uint8Array) {
  if (a.length !== b.length) return false;
  let result = 0;
  for (let i = 0; i < a.length; i++) result |= (a[i] ?? 0) ^ (b[i] ?? 0);
  return result === 0;
}

export class SessionService extends Effect.Service<SessionService>()("SessionService", {
  effect: Effect.gen(function* () {
    const rootDb = yield* PgRootDB;

    return {
      createSession: (userId: string) =>
        Effect.gen(function* () {
          const user = yield* rootDb
            .selectFrom("app_public.users")
            .select(["id", "username", "role", "email", "name"])
            .where("id", "=", userId)
            .pipe(Effect.head, Effect.catchTag("NoSuchElementException", () =>
              Effect.die(new Error(`User ${userId} not found`)),
            ));

          const id = generateSecureRandomString();
          const secret = generateSecureRandomString();
          const secretHash = hashSecret(secret).toString("hex");
          const sessionUuid = crypto.randomUUID();
          const expiresAt = new Date(Date.now() + SESSION_TTL_MS);

          const key = sessionKey(id);
          yield* Effect.tryPromise(() =>
            valkey.hset(key, {
              secret_hash: secretHash,
              session_uuid: sessionUuid,
              user_id: user.id,
              username: user.username,
              role: user.role,
            }),
          );
          yield* Effect.tryPromise(() => valkey.expire(key, SESSION_TTL_SECONDS));

          yield* rootDb.insertInto("app_private.sessions").values({
            id: sessionUuid,
            user_id: user.id,
            expires_at: expiresAt,
          });

          return { token: `${id}.${secret}`, expiresAt };
        }),

      validateSessionToken: (token?: string) =>
        Effect.gen(function* () {
          const noSession = { user: null, session: null } as const;
          if (!token) return noSession;

          const [id, secret] = token.split(".");
          if (!id || !secret) return noSession;

          const data = yield* Effect.tryPromise(() => valkey.hgetall(sessionKey(id)));
          if (!data?.secret_hash) return noSession;

          const incomingHash = hashSecret(secret);
          const storedHash = Buffer.from(data.secret_hash, "hex");
          if (!constantTimeEqual(incomingHash, storedHash)) return noSession;

          return {
            user: {
              id: data.user_id!,
              username: data.username!,
              role: data.role! as "admin" | "student",
            },
            session: {
              id: data.session_uuid!,
              cookie_id: id,
              user_id: data.user_id!,
            },
          } as const;
        }),

      deleteSession: (cookieId: string) =>
        Effect.gen(function* () {
          const key = sessionKey(cookieId);
          const sessionUuid = yield* Effect.tryPromise(() => valkey.hget(key, "session_uuid"));
          yield* Effect.tryPromise(() => valkey.del(key));
          if (sessionUuid) {
            yield* rootDb.deleteFrom("app_private.sessions").where("id", "=", sessionUuid);
          }
        }),
    } as const;
  }),
  dependencies: [PgRootDB.Live],
  accessors: true,
}) {}
```

```typescript
      // CORS preflight
      if (request.method === "OPTIONS") {
        return new Response(null, { status: 204, headers: corsHeaders(origin) });
      }

      // API routes
      if (isApiPath(url.pathname)) {
        const response = await apiWebHandler.handler(request);
        const cors = corsHeaders(origin);
        for (const [k, v] of Object.entries(cors)) {
          response.headers.set(k, v);
        }
        return response;
      }

      // File download (special route handled outside HttpApi for streaming)
      if (url.pathname.startsWith("/api/materials/") && url.pathname.endsWith("/download")) {
        // TODO: implement file streaming with auth check
      }

      // Static assets from dist/
      if (indexHtml) {
        const assetPath = join(distDir, url.pathname);
        if (url.pathname !== "/" && existsSync(assetPath)) {
          return new Response(Bun.file(assetPath));
        }
        // SPA fallback
        return new Response(indexHtml, {
          headers: { "Content-Type": "text/html" },
        });
      }

      // Dev mode: proxy to Vite
      return new Response("SPA not built. Run `bun run build` or use Vite dev server.", { status: 404 });
    },
  });

  console.log(`Server running at http://localhost:${PORT}`);

  const shutdown = (signal: string) => {
    console.log(`\n${signal} received, shutting down...`);
    Promise.all([apiWebHandler.dispose()]).then(() => {
      server.stop();
      process.exit(0);
    });
  };

  process.on("SIGINT", () => shutdown("SIGINT"));
  process.on("SIGTERM", () => shutdown("SIGTERM"));
}
```

Simple fetch-based API client as an Effect Service.

```typescript
import { Effect, Context } from "effect";

type RequestOpts = {
  method?: string;
  body?: unknown;
  params?: Record<string, string | number>;
};

async function apiFetch<T>(path: string, opts: RequestOpts = {}): Promise<T> {
  const url = new URL(path, window.location.origin);
  if (opts.params) {
    for (const [k, v] of Object.entries(opts.params)) {
      url.searchParams.set(k, String(v));
    }
  }
  const res = await fetch(url, {
    method: opts.method ?? "GET",
    headers: {
      ...(opts.body ? { "Content-Type": "application/json" } : {}),
      "X-Requested-With": "fibrae",
    },
    credentials: "include",
    body: opts.body ? JSON.stringify(opts.body) : undefined,
  });
  if (!res.ok) {
    const err = await res.json().catch(() => ({ error: res.statusText }));
    throw new Error(err.error ?? `HTTP ${res.status}`);
  }
  return res.json();
}

export class ApiClient extends Context.Tag("ApiClient")<ApiClient, {
  get: <T>(path: string, params?: Record<string, string | number>) => Effect.Effect<T>;
  post: <T>(path: string, body?: unknown) => Effect.Effect<T>;
  patch: <T>(path: string, body?: unknown) => Effect.Effect<T>;
  del: <T>(path: string) => Effect.Effect<T>;
}>() {}

export const ApiClientLive = ApiClient.of({
  get: <T>(path: string, params?: Record<string, string | number>) =>
    Effect.tryPromise(() => apiFetch<T>(path, { params })),
  post: <T>(path: string, body?: unknown) =>
    Effect.tryPromise(() => apiFetch<T>(path, { method: "POST", body })),
  patch: <T>(path: string, body?: unknown) =>
    Effect.tryPromise(() => apiFetch<T>(path, { method: "PATCH", body })),
  del: <T>(path: string) =>
    Effect.tryPromise(() => apiFetch<T>(path, { method: "DELETE" })),
});
```
